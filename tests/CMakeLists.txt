cmake_minimum_required(VERSION 3.22)
project(voxen_scene_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(test_name IN ITEMS scene_ids_smoke scene_repository_smoke runtime_snapshot_smoke floor_portal_placement_smoke)
    add_executable(${test_name} ${CMAKE_CURRENT_SOURCE_DIR}/${test_name}.cpp)
    target_include_directories(${test_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../engine/scene/include)
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
